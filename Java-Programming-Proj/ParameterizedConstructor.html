class demonstration
{ int x;
int y;
float z;
demonstration(int x1,int y1,int z1)
{
x=x1;
y=y1;
z=z1;
}
void display()
{
System.out.println("Values of x, y and z are:"+x+" "+y+" "+z);
}
}
class Demomain1
{ public static void main(String args[])
{ demonstration d1=new demonstration(1,2,3); // this is a call for the above parameterized constructor
d1.display();
}
}

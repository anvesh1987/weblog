class Student2 {
int id;
String name;
double marks;
// Default Constructor
Student2() {
id = 0;
name = "Unknown";
marks = 0.0;
}
// Constructor with two parameters
Student2(int i, String n) {
id = i;
name = n;
marks = 0.0;
}
// Constructor with three parameters
Student2(int i, String n, double m) {
id = i;
name = n;
marks = m;
}
// Method to display student details
void display() {
System.out.println("ID : " + id);
System.out.println("Name : " + name);
System.out.println("Marks : " + marks);
System.out.println();
}
}
public class ConstructorOverloadingDemo2 {
public static void main(String[] args) {
// Calls the default constructor
Student2 s1 = new Student2();
// Calls the constructor with two parameters
Student2 s2 = new Student2(101, "ABC");
// Calls the constructor with three parameters
Student2 s3 = new Student2(102, "DEF", 92.5);
s1.display();
s2.display();
s3.display();
}
}
